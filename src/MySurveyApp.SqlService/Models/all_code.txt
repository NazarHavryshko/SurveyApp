﻿using Microsoft.EntityFrameworkCore;
using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;
using MySurveyApp.Shared.Enums;

namespace MySurveyApp.SqlService.Models
{
    
    [Table("survey")]
    public class Survey
    {
        [Key]
        [DatabaseGenerated(DatabaseGeneratedOption.Identity)]
        [Column("survey_id")]
        public int SurveyId { get; set; }

        [Required]
        [MaxLength(500)]
        [Column("survey_text ")]
        public string SurveyText { get; set; } = string.Empty;

        [Required]
        [Column("survey_type")]
        public SurveyType SurveyType { get; set; }

        [Required]
        [Column("region_id")]
        public int? RegionId { get; set; }

        [Required]
        [Column("topic_id")]
        public int TopicId { get; set; }

        [Required]
        [Column("author_id")]
        public Guid AuthorId { get; set; }

        [Required]
        [Column("status")]
        public SurveyStatus Status { get; set; }  = SurveyStatus.Pending;

        [Required]
        [Column("created_at")]
        public DateTime CreatedAt { get; set; } = DateTime.UtcNow;

        [Required]
        [Column("expires_at")]
        public DateTime ExpiresAt { get; set; }


        [ForeignKey(nameof(RegionId))]
        public virtual Region? Region { get; set; }

        [ForeignKey(nameof(TopicId))]
        public virtual SurveyTopic? SurveyTopic { get; set; }

        [ForeignKey(nameof(AuthorId))]
        public virtual User? Author { get; set; }

    }
}

﻿using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;
using MySurveyApp.Shared.Enums;

namespace MySurveyApp.SqlService.Models
{
    [Table("users")]
    public class User
    {
        [Key]
        [Column("user_id")]
        public Guid UserId { get; set; } = Guid.NewGuid();

        [Required]
        [Column("user_type")]
        public UserType UserType { get; set; }  = UserType.Anonymous;

        [MaxLength(100)]
        [Column("nickname")]
        public string? Nickname { get; set; }

        [MaxLength(255)]
        [Column("password_hash")]
        public string? PasswordHash { get; set; }

        [MaxLength(10)]
        [Column("gender")]
        public GenderType? Gender { get; set; }

        [Column("birth_year")]
        public int? BirthYear { get; set; }

        [Column("occupation_id")]
        public int? OccupationId { get; set; }

        [Column("region_id")]
        public int? RegionId { get; set; }

        [Column("hobby_id")]
        public int? HobbyId { get; set; }

        [Column("answers_count")]
        public int AnswersCount { get; set; } = 0;

        [Column("available_question_tokens")]
        public int AvailableQuestionTokens { get; set; } = 0;



        [ForeignKey(nameof(OccupationId))]
        public virtual Occupation? Occupation { get; set; }

        [ForeignKey(nameof(HobbyId))]
        public virtual Hobby? Hobby { get; set; }

        [ForeignKey(nameof(RegionId))]
        public virtual Region? Region { get; set; }

    }
}
