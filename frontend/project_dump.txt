function fnv1aHash(str: string, seed: number): string {
  let h = seed >>> 0;
  for (let i = 0; i < str.length; i++) {
    h ^= str.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return (h >>> 0).toString(16).padStart(8, '0');
}

/**
 * РџРѕРІРµСЂС‚Р°С” РїРѕСЃС‚С–Р№РЅРёР№ СѓРЅС–РєР°Р»СЊРЅРёР№ UUID РїСЂРёСЃС‚СЂРѕСЋ/Р±СЂР°СѓР·РµСЂР°, СЏРєРёР№ РЅРµ Р·РјС–РЅСЋС”С‚СЊСЃСЏ.

export function getPersistentDeviceId(): string {
  const cached = localStorage.getItem(DEVICE_ID_KEY);
  if (cached) return cached;

  const signature = [
    navigator.userAgent,
    navigator.language,
    `${screen.width}x${screen.height}`,
    screen.colorDepth,
    Intl.DateTimeFormat().resolvedOptions().timeZone,
    navigator.hardwareConcurrency ?? '0',
    navigator.platform ?? '',
  ].join('|');

  const p1 = fnv1aHash(signature, 0x811c9dc5);
  const p2 = fnv1aHash(signature, 0x01000193);
  const p3 = fnv1aHash(signature, 0x1b873593);
  const p4 = fnv1aHash(signature, 0x85ebca6b);

  // Р¤РѕСЂРјР°С‚ РІР°Р»С–РґРЅРѕРіРѕ UUIDv4 РґР»СЏ СЃСѓРјС–СЃРЅРѕСЃС‚С– Р· Guid Сѓ C#
  const deviceUuid = `${p1}-${p2.slice(0, 4)}-4${p2.slice(5, 8)}-a${p3.slice(1, 4)}-${p3.slice(4, 8)}${p4}`;

  localStorage.setItem(DEVICE_ID_KEY, deviceUuid);
  return deviceUuid;
}

export const apiClient = axios.create({
  baseURL: '/api',
  headers: {

apiClient.interceptors.request.use((config) => {
  const jwt = localStorage.getItem(JWT_TOKEN_KEY);
  const anonToken = localStorage.getItem(ANON_TOKEN_KEY) || getPersistentDeviceId();

  if (jwt) {
    config.headers.Authorization = `Bearer ${jwt}`;
  }
  config.headers['X-Anonymous-Token'] = anonToken;

  return config;
});

/**
 * РџРµСЂРµРІС–СЂСЏС”, С‡Рё РІР¶Рµ СЃС‚РІРѕСЂРµРЅРѕ Р°РЅРѕРЅС–РјРЅРёР№ РїСЂРѕС„С–Р»СЊ РЅР° РїСЂРёСЃС‚СЂРѕС—.

export async function ensureAnonymousSession(): Promise<{
  deviceId: string;
  isRegistered: boolean;
  fromCache: boolean;
}> {
  const deviceId = getPersistentDeviceId();
  const hasJwt = Boolean(localStorage.getItem(JWT_TOKEN_KEY));
  const alreadyRegistered = localStorage.getItem(ANON_REGISTERED_KEY) === 'true';

  if (hasJwt || alreadyRegistered) {
    if (!localStorage.getItem(ANON_TOKEN_KEY)) {
      localStorage.setItem(ANON_TOKEN_KEY, deviceId);
    }
    return { deviceId, isRegistered: true, fromCache: true };
  }

  try {
    const res = await apiClient.post('/auth/anonymous', {
      deviceId,
      anonymousToken: deviceId,
    });

    const returnedToken =
      res.data?.anonymousToken || res.data?.userId || res.data?.token || deviceId;

    localStorage.setItem(ANON_TOKEN_KEY, String(returnedToken));
    localStorage.setItem(ANON_REGISTERED_KEY, 'true');

    return { deviceId, isRegistered: true, fromCache: false };
  } catch {
    // Р—Р±РµСЂС–РіР°С”РјРѕ Р»РѕРєР°Р»СЊРЅРёР№ С‚РѕРєРµРЅ РїСЂРёСЃС‚СЂРѕСЋ, С‰РѕР± РЅРµ Р±Р»РѕРєСѓРІР°С‚Рё РєРѕСЂРёСЃС‚СѓРІР°С‡Р°
    localStorage.setItem(ANON_TOKEN_KEY, deviceId);
    return { deviceId, isRegistered: false, fromCache: false };
  }
}
